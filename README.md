<?php

$товары = [
    "Хлеб" => 60,
    "Молоко" => 90,
    "Сыр" => 200,
    "Вода" => 40
];

$покупатели = [
    "влад" => ["Хлеб", "Молоко", "Вода"],
    "кира" => ["Молоко", "Сыр"],
    "тимофей" => ["Хлеб", "Сыр", "Вода"],
];

foreach ($покупатели as $name => $korzina) {
    echo "Покупатель: $name<br>";
    echo "Купил:<br>";

    $itog = 0;

    foreach ($korzina as $tovar) {
        $price = $товары[$tovar];
        echo "- $tovar ($price руб.)<br>";
        $itog += $price;
    }

    echo "Итого к оплате: $itog руб.<br><br>";
}
